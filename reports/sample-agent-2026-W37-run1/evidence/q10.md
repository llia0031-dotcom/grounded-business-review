```sql
SELECT o.pay_week, o.merchant_type, COUNT(*) AS n, ROUND(SUM(f.refund_amount),2) AS amt, COUNT(DISTINCT o.merchant) AS merchants, MIN(CAST(f.refund_date AS VARCHAR)) AS d_min, MAX(CAST(f.refund_date AS VARCHAR)) AS d_max FROM fact_refund f JOIN order_wide o USING(order_id) WHERE o.category='美发美甲' AND f.refund_type='主动退' AND CAST(f.refund_date AS VARCHAR) BETWEEN '2026-09-07' AND '2026-09-13' GROUP BY 1,2 ORDER BY 1,2
```
